## 素材再生命令（换素材时照此重跑）

```bash
# 视频（两幅 SVG 交叉淡入淡出 → 320×180 的 mp4 与 webm）
ffmpeg -loop 1 -t 1.3 -i site/assets/img/demo-wide.svg -loop 1 -t 1.3 -i site/assets/img/demo-landscape.svg \
  -filter_complex "[0:v]scale=320:180:force_original_aspect_ratio=decrease,pad=320:180:(ow-iw)/2:(oh-ih)/2:color=faf6ec,fps=12,format=yuv420p[v0];\
[1:v]scale=320:180:force_original_aspect_ratio=decrease,pad=320:180:(ow-iw)/2:(oh-ih)/2:color=faf6ec,fps=12,format=yuv420p[v1];\
[v0][v1]xfade=transition=fade:duration=0.5:offset=0.8,format=yuv420p[v]" \
  -map "[v]" -c:v libx264 -profile:v baseline -level 3.0 -pix_fmt yuv420p -crf 32 -preset veryfast \
  -movflags +faststart site/assets/media/demo-fade.mp4
ffmpeg -i site/assets/media/demo-fade.mp4 -c:v libvpx-vp9 -crf 40 -b:v 0 site/assets/media/demo-fade.webm
# 音频（1.8 秒纯音，音量压到 0.18，带淡出）
ffmpeg -f lavfi -i "sine=frequency=587.33:duration=1.8" \
  -af "volume=0.18,afade=t=out:st=1.1:d=0.7" -c:a libmp3lame -b:a 64k site/assets/media/demo-tone.mp3
```
